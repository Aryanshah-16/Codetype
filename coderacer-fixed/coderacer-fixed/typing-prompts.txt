public int findMax(int[] arr) {

    for (int n : arr) {
        if (n > max) max = n;
    }
    return max;
}

public int countVowels(String s) {
    int count = 0;
    for (char c : s.toCharArray()) {
        if ("aeiou".indexOf(c) >= 0) count++;
    }
    return count;
}
